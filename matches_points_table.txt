CREATE TABLE matches_table(
    match_no INT,
    team_a   VARCHAR(10),
    team_b   VARCHAR(10),
    winner   VARCHAR(10)
);

team	wins	losses	draws

WITH data AS (
    SELECT 
        team_a AS team, 
        CASE WHEN team_a = winner THEN 1 ELSE 0 END AS win_flag,
        CASE WHEN team_b = winner THEN 1 ELSE 0 END AS loss_flag,
        CASE WHEN winner = 'Draw' THEN 1 ELSE 0 END AS draw_flag
    FROM matches_table
    UNION ALL
    SELECT 
        team_b, 
        CASE WHEN team_b = winner THEN 1 ELSE 0 END AS win_flag,
        CASE WHEN team_a = winner THEN 1 ELSE 0 END AS loss_flag,
        CASE WHEN winner = 'Draw' THEN 1 ELSE 0 END AS draw_flag
    FROM matches_table
)
SELECT 
    team, 
    SUM(win_flag) AS wins, 
    SUM(loss_flag) AS losses, 
    SUM(draw_flag) AS draws
FROM data
GROUP BY team;
